A.1 Black–Scholes, Greeks and implied volatility

# Black-Scholes price, Greeks and implied volatility (Chapters 9-10)
import numpy as np
from scipy.stats import norm
from scipy.optimize import brentq

def bs(S, K, r, sig, T, kind="call"):
    d1 = (np.log(S/K) + (r + 0.5*sig**2)*T) / (sig*np.sqrt(T)); d2 = d1 - sig*np.sqrt(T)
    if kind == "call": return S*norm.cdf(d1) - K*np.exp(-r*T)*norm.cdf(d2)
    return K*np.exp(-r*T)*norm.cdf(-d2) - S*norm.cdf(-d1)

def greeks(S, K, r, sig, T):
    d1 = (np.log(S/K) + (r + 0.5*sig**2)*T) / (sig*np.sqrt(T)); d2 = d1 - sig*np.sqrt(T)
    return dict(delta=norm.cdf(d1), gamma=norm.pdf(d1)/(S*sig*np.sqrt(T)),
                vega=S*norm.pdf(d1)*np.sqrt(T),
                theta=-S*norm.pdf(d1)*sig/(2*np.sqrt(T)) - r*K*np.exp(-r*T)*norm.cdf(d2))

def implied_vol(price, S, K, r, T, kind="call"):
    return brentq(lambda s: bs(S, K, r, s, T, kind) - price, 1e-4, 5.0)

print(bs(100, 100, 0.05, 0.20, 1.0))            # 10.4506
print(implied_vol(10.4506, 100, 100, 0.05, 1.0)) # 0.2000

A.2 Monte Carlo with antithetic and control variates; Heston by full-truncation Euler

# Risk-neutral Monte Carlo for a call (Chapters 9, 11, 13)
rng = np.random.default_rng(0)
S0, K, r, sig, T, M = 100., 100., 0.05, 0.20, 1.0, 100_000
Z  = rng.normal(size=M)
ST = S0*np.exp((r - 0.5*sig**2)*T + sig*np.sqrt(T)*Z)        # exact GBM, drift r not mu
pay = np.exp(-r*T)*np.maximum(ST - K, 0)
print(pay.mean(), pay.std()/np.sqrt(M))                        # plain: price +- se

STa = S0*np.exp((r - 0.5*sig**2)*T - sig*np.sqrt(T)*Z)       # antithetic pair
pay_a = 0.5*(pay + np.exp(-r*T)*np.maximum(STa - K, 0))

cv = np.exp(-r*T)*ST                                           # control variate, E[cv] = S0
beta = np.cov(pay, cv)[0, 1]/np.var(cv)
pay_c = pay - beta*(cv - S0)
print(pay_c.mean(), pay_c.std()/np.sqrt(M))                    # ~7x smaller variance

# Heston: dv = kappa(theta - v)dt + xi sqrt(v) dW2, corr(dW1, dW2) = rho
v0, kappa, theta, xi, rho, n = 0.04, 2.0, 0.04, 0.39, -0.7, 500
dt = T/n; S = np.full(M, S0); v = np.full(M, v0)
for _ in range(n):
    z1 = rng.normal(size=M); z2 = rho*z1 + np.sqrt(1 - rho**2)*rng.normal(size=M)
    vp = np.maximum(v, 0)                                      # full truncation keeps v >= 0
    S = S*np.exp((r - 0.5*vp)*dt + np.sqrt(vp*dt)*z1)
    v = v + kappa*(theta - vp)*dt + xi*np.sqrt(vp*dt)*z2
print(np.exp(-r*T)*np.maximum(S - K, 0).mean())                # ~10.29

A.3 Crank–Nicolson for the Black–Scholes PDE

# Crank-Nicolson on a uniform S grid, European call (Chapters 5, 13)
from scipy.linalg import solve_banded

def cn_call(S0, K, r, sig, T, nS=400, nT=400, Smax=300.0):
    S = np.linspace(0, Smax, nS + 1); dt = T/nT
    V = np.maximum(S - K, 0)                                  # payoff = initial condition in tau
    i = np.arange(1, nS)
    a = 0.25*dt*(sig**2*i**2 - r*i); b = -0.5*dt*(sig**2*i**2 + r); c = 0.25*dt*(sig**2*i**2 + r*i)
    ab = np.zeros((3, nS - 1)); ab[0, 1:] = -c[:-1]; ab[1] = 1 - b; ab[2, :-1] = -a[1:]
    for k in range(nT):
        tau = (k + 1)*dt
        rhs = a*V[:-2] + (1 + b)*V[1:-1] + c*V[2:]
        rhs[-1] += c[-1]*(Smax - K*np.exp(-r*tau))            # upper boundary: V ~ S - K e^{-r tau}
        Vn = np.zeros_like(V); Vn[-1] = Smax - K*np.exp(-r*tau)
        Vn[1:-1] = solve_banded((1, 1), ab, rhs); V = Vn
    return np.interp(S0, S, V)

print(cn_call(100, 100, 0.05, 0.20, 1.0))                     # 10.4515

A.4 GARCH(1,1) by maximum likelihood

# GARCH(1,1): simulate, then fit by MLE (Chapter 15)
from scipy.optimize import minimize

def garch_nll(p, x):
    om, al, be = p
    if om <= 0 or al < 0 or be < 0 or al + be >= 1: return 1e10
    h = np.empty(len(x)); h[0] = x.var()
    for t in range(1, len(x)): h[t] = om + al*x[t-1]**2 + be*h[t-1]
    return 0.5*np.sum(np.log(h) + x**2/h)                # -log L up to a constant

om, al, be, n = 2e-6, 0.08, 0.90, 3000
x = np.zeros(n); h = np.zeros(n); h[0] = om/(1 - al - be)
for t in range(1, n):
    h[t] = om + al*x[t-1]**2 + be*h[t-1]; x[t] = np.sqrt(h[t])*rng.normal()
fit = minimize(garch_nll, [1e-6, 0.05, 0.9], args=(x,), method="Nelder-Mead")
print(fit.x, fit.x[1] + fit.x[2])                              # omega, alpha, beta, persistence

A.5 Marchenko–Pastur eigenvalue clipping

# Clean a sample correlation matrix by clipping the noise bulk (Chapter 16)
def clip_correlation(R):
    """R: T x N matrix of returns. Returns the cleaned covariance."""
    T, N = R.shape; q = N/T; lam_plus = (1 + np.sqrt(q))**2
    S = np.cov(R.T); sd = np.sqrt(np.diag(S)); C = S/np.outer(sd, sd)
    ev, EV = np.linalg.eigh(C)
    noise = ev < lam_plus
    ev[noise] = ev[noise].mean()                               # keep the trace, flatten the bulk
    Cc = EV @ np.diag(ev) @ EV.T
    d = np.sqrt(np.diag(Cc)); Cc = Cc/np.outer(d, d)           # restore unit diagonal
    return Cc*np.outer(sd, sd)

def min_variance(Sigma):
    w = np.linalg.solve(Sigma, np.ones(len(Sigma))); return w/w.sum()

A.6 Longstaff–Schwartz for an American put

# LSMC: regression-based continuation values (Chapter 13)
M, n = 100_000, 50; dt = T/n
S = np.empty((M, n + 1)); S[:, 0] = S0
for k in range(n):
    S[:, k+1] = S[:, k]*np.exp((r - 0.5*sig**2)*dt + sig*np.sqrt(dt)*rng.normal(size=M))
cash = np.maximum(K - S[:, -1], 0)
for k in range(n - 1, 0, -1):
    cash *= np.exp(-r*dt)
    itm = K - S[:, k] > 0                                      # regress only in-the-money paths
    X = S[itm, k]; A = np.column_stack([np.ones_like(X), X, X**2, X**3])
    coef = np.linalg.lstsq(A, cash[itm], rcond=None)[0]
    cont = A @ coef; ex = K - X
    idx = np.where(itm)[0][ex > cont]; cash[idx] = ex[ex > cont]
print(np.exp(-r*dt)*cash.mean())                               # ~6.08 (tree: 6.09; European: 5.57)
